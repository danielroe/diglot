---
title: Translate an Angular app
description: Apply diglot bundles to Markdown documents in an Angular app, in the browser and while it prerenders.
---

diglot doesn't need an Angular package: an Angular app that renders Markdown can load a bundle and apply it with `@diglot/mdast`. This recipe puts that in one service and one function. The store, the document function and the language file below are the ones in the [Angular app example](/guide/examples/angular-app), which you can run as it is.

## Prepare the translations repository

Your app applies translations at runtime, so it reads the bundles that [`diglot build`](/workflow/cli/build) writes and the policy snapshot that [`diglot extract`](/workflow/cli/extract) writes. Run both before `ng build`:

```json [package.json]
{
  "scripts": {
    "translations:build": "diglot extract --cwd translations && diglot build --cwd translations",
    "prestart": "pnpm translations:build",
    "start": "ng serve",
    "prebuild": "pnpm translations:build",
    "build": "ng build"
  }
}
```

Set `parser` to `mdast` in the translations repository, because the app applies translations with the same parser. `satteri` needs native bindings that browsers can't load.

```ts [translations/diglot.config.ts]
import { defineConfig } from '@diglot/core'

export default defineConfig({
  locales: ['fr', 'tr'],
  parser: 'mdast',
})
```

Serve the bundles and your Markdown files with your app, and add `satteri` to `externalDependencies`. The mdast adapter imports it only when its parser is `satteri`, but the Angular build tries to bundle that import all the same.

```json [angular.json]
{
  "externalDependencies": ["satteri"],
  "assets": [
    { "glob": "*.json", "input": "translations/dist", "output": "translations" },
    { "glob": "**/*.md", "input": "src/content/docs", "output": "docs" }
  ]
}
```

Install `@diglot/core`, `@diglot/mdast` and `@diglot/store`, and `@diglot/cli` as a dev dependency. Then add these providers to your application config:

```ts [src/app/app.config.ts]
import type { ApplicationConfig } from '@angular/core'
import { provideHttpClient } from '@angular/common/http'
import { provideClientHydration } from '@angular/platform-browser'
import { provideRouter, withComponentInputBinding } from '@angular/router'
import { routes } from './app.routes'

export const appConfig: ApplicationConfig = {
  providers: [
    provideRouter(routes, withComponentInputBinding()),
    provideHttpClient(),
    provideClientHydration(),
  ],
}
```

`withComponentInputBinding()` passes route parameters to component inputs, and `provideClientHydration()` reuses the prerendered page in the browser.

## Load and apply a document

The store fetches each bundle once and applies it to a document. Its `./locales` import is a file of your own that exports `defaultLocale`, the language your documents are written in, such as `'en'`. On the server, it puts the translated document in `TransferState`, so the browser shows it while it hydrates without fetching anything. Once the app is stable, documents load from the network.

```ts [src/app/diglot-store.ts]
import type { ApplyReport, PolicySnapshot, TranslationRecord } from '@diglot/core'
import { isPlatformBrowser, isPlatformServer } from '@angular/common'
import { HttpClient } from '@angular/common/http'
import { ApplicationRef, inject, Injectable, makeStateKey, PLATFORM_ID, TransferState } from '@angular/core'
import { restorePolicy } from '@diglot/core'
import { parseBundle } from '@diglot/store/runtime'
import { firstValueFrom } from 'rxjs'
import snapshot from '../../translations/.diglot/policy.json'
import { defaultLocale } from './locales'

export interface DocumentRequest {
  locale: string
  path: string
}

export interface TranslatedDocument {
  markdown: string
  report: ApplyReport | null
}

const policy = restorePolicy(snapshot as PolicySnapshot)

@Injectable({ providedIn: 'root' })
export class DiglotStore {
  private readonly http = inject(HttpClient)
  private readonly transferState = inject(TransferState)
  private readonly server = isPlatformServer(inject(PLATFORM_ID))
  private readonly bundles = new Map<string, Promise<ReadonlyMap<string, TranslationRecord>>>()
  private hydrating = isPlatformBrowser(inject(PLATFORM_ID))

  constructor() {
    if (this.hydrating)
      void inject(ApplicationRef).whenStable().then(() => this.hydrating = false)
  }

  transferred({ locale, path }: DocumentRequest): TranslatedDocument | undefined {
    return this.hydrating ? this.transferState.get(stateKey(locale, path), undefined) : undefined
  }

  async load({ locale, path }: DocumentRequest): Promise<TranslatedDocument> {
    const document = locale === defaultLocale
      ? { markdown: await this.fetchText(sourceUrl(path)), report: null }
      : await this.translate(locale, path)
    if (this.server)
      this.transferState.set(stateKey(locale, path), document)
    return document
  }

  private async translate(locale: string, path: string): Promise<TranslatedDocument> {
    const [source, records, { apply }] = await Promise.all([
      this.fetchText(sourceUrl(path)),
      this.bundle(locale),
      import('@diglot/mdast'),
    ])
    const { markdown, report } = await apply(source, records, { policy, file: `src/content/docs/${path}.md` })
    return { markdown, report }
  }

  private bundle(locale: string): Promise<ReadonlyMap<string, TranslationRecord>> {
    let bundle = this.bundles.get(locale)
    if (!bundle) {
      bundle = this.fetchText(`/translations/${locale}.json`).then(parseBundle)
      bundle.catch(() => this.bundles.delete(locale))
      this.bundles.set(locale, bundle)
    }
    return bundle
  }

  private fetchText(url: string): Promise<string> {
    return firstValueFrom(this.http.get(url, { responseType: 'text', transferCache: false }))
  }
}

function sourceUrl(path: string): string {
  return `/docs/${path}.md`
}

function stateKey(locale: string, path: string) {
  return makeStateKey<TranslatedDocument>(`diglot:${locale}:${path}`)
}
```

Records are matched to your documents by hashes, which only agree under the policy the records were extracted with, so the store restores the snapshot in `.diglot/policy.json`. `restorePolicy` throws when the snapshot is invalid. If your policy has a custom function, pass it as the second argument. See [snapshots](/core/policy-and-config#snapshots). `file` is the path that `diglot extract` saw, which a custom `salt` can read.

## Read a document in a component

`diglotDocument()` turns the store into signals. While the next document loads, it keeps the previous one, so a page doesn't go blank while its component stays on screen.

```ts [src/app/diglot-document.ts]
import type { DocumentRequest, TranslatedDocument } from './diglot-store'
import { computed, inject, linkedSignal, resource } from '@angular/core'
import { DiglotStore } from './diglot-store'
import { DIGLOT_LOCALE } from './language'

interface Shown {
  request: DocumentRequest
  document: TranslatedDocument
}

export function diglotDocument(path: () => string) {
  const store = inject(DiglotStore)
  const locale = inject(DIGLOT_LOCALE)
  const request = computed<DocumentRequest>(() => ({ locale: locale(), path: path() }), {
    equal: (a, b) => a.locale === b.locale && a.path === b.path,
  })
  const transferred = computed(() => store.transferred(request()))
  const loaded = resource({
    params: () => transferred() ? undefined : request(),
    loader: ({ params }) => store.load(params).then((document): Shown => ({ request: params, document })),
  })
  const shown = linkedSignal<{ next: Shown | undefined, failed: boolean }, Shown | undefined>({
    source: () => {
      const document = transferred()
      return {
        next: document ? { request: request(), document } : loaded.hasValue() ? loaded.value() : undefined,
        failed: loaded.error() !== undefined,
      }
    },
    computation: ({ next, failed }, previous) => next ?? (failed ? undefined : previous?.value),
  })
  return {
    locale: computed(() => shown()?.request.locale ?? request().locale),
    markdown: computed(() => shown()?.document.markdown),
    report: computed(() => shown()?.document.report),
    isLoading: loaded.isLoading,
    error: computed(() => loaded.error()),
    reload: () => loaded.reload(),
  }
}
```

Call it in a field initialiser, and render the Markdown with whatever renderer you already use. A minimal page looks like this:

```ts
import { Component, computed, input } from '@angular/core'
import { marked } from 'marked'
import { diglotDocument } from './diglot-document'

@Component({
  selector: 'app-page',
  templateUrl: './page.html',
})
export class Page {
  readonly slug = input.required<string>()
  protected readonly doc = diglotDocument(() => this.slug())
  protected readonly partlyTranslated = computed(() => {
    const report = this.doc.report()
    return !!report && (report.missing > 0 || report.rejected.length > 0)
  })

  protected readonly html = computed(() => {
    const body = this.doc.markdown()?.replace(/^---\n[\s\S]*?\n---\n/, '') ?? ''
    return marked.parse(body, { async: false })
  })
}
```

```html
@if (partlyTranslated()) {
  <aside>This page is only partly translated.</aside>
}
<article [innerHTML]="html()"></article>
```

`markdown` includes the frontmatter, so remove or read it before you render the body. `report` is the [report](/core/apply) of `apply`, and `null` for the original language. Text without a usable translation stays in the source language. After an `error`, `reload()` loads the document again, and a bundle that failed to load is fetched again. It does nothing for a document that came from the server.

## Choose the language

`DIGLOT_LOCALE` holds a function that returns the current language. You can provide it from the URL or from a signal, for a whole app or for the `providers` of a route.

```ts [src/app/language.ts]
import type { Provider } from '@angular/core'
import type { ActivatedRouteSnapshot } from '@angular/router'
import { inject, InjectionToken, signal } from '@angular/core'
import { toSignal } from '@angular/core/rxjs-interop'
import { NavigationEnd, Router } from '@angular/router'
import { filter, map } from 'rxjs'
import { defaultLocale } from './locales'

export const DIGLOT_LOCALE = new InjectionToken<() => string>('DIGLOT_LOCALE')

export const lang = signal<string>(defaultLocale)

export function provideRouteLocale(param = 'locale'): Provider {
  return {
    provide: DIGLOT_LOCALE,
    useFactory: () => {
      const router = inject(Router)
      const read = () => routeParam(router.routerState.snapshot.root, param) ?? defaultLocale
      return toSignal(router.events.pipe(filter(event => event instanceof NavigationEnd), map(read)), { initialValue: read() })
    },
  }
}

function routeParam(root: ActivatedRouteSnapshot, param: string): string | undefined {
  let value: string | undefined
  for (let route: ActivatedRouteSnapshot | null = root; route; route = route.firstChild)
    value = route.params[param] ?? value
  return value
}
```

Add `provideRouteLocale()` to the routes with a language parameter, so `/fr/guide` is French. Each language has its own link, and each one can be prerendered.

```ts
export const routes: Routes = [
  { path: ':locale', providers: [provideRouteLocale()], children: [{ path: ':slug', component: Page }] },
]
```

To keep the language out of the URL, provide a signal and set it from a language menu:

```ts
export const routes: Routes = [
  { path: ':slug', component: Page, providers: [{ provide: DIGLOT_LOCALE, useValue: lang }] },
]
```

```html
<button type="button" (click)="lang.set('fr')">Français</button>
```

The component that renders the button needs the signal as a field, `protected readonly lang = lang`. Documents follow the signal without a navigation. Only the starting language can be prerendered, and a reload starts over unless you store the choice yourself. Keep the signal for choices made in the browser: a module-level value is shared between requests on a live server.

## Prerender the pages

Set up server rendering with `ng add @angular/ssr`, then give every language and page to your server routes:

```ts
export const serverRoutes: ServerRoute[] = [
  {
    path: ':locale/:slug',
    renderMode: RenderMode.Prerender,
    getPrerenderParams: async () => locales.flatMap(locale => pages.map(slug => ({ locale, slug }))),
  },
]
```

The server waits for each document, and the build reads your bundles and documents from its own output.

## Development

During `ng serve`, a change to a source document shows up when you reload the page. Its edited segments are in the source language until their records are updated, because their hashes changed. After you change a record or `diglot.config.ts`, run `translations:build` again and reload the page.
